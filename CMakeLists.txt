cmake_minimum_required(VERSION 3.16)
project(anc216 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(ANC216_LLVM_LLC "" CACHE FILEPATH "Optional ANC216-enabled llc for compiler integration tests")
add_subdirectory(assembler)
add_subdirectory(disassembler)
add_subdirectory(emulator)
add_subdirectory(cardreader)
if(BUILD_TESTING AND ANC216_LLVM_LLC)
    get_filename_component(ANC216_LLVM_BIN "${ANC216_LLVM_LLC}" DIRECTORY)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(llvm_guest_runner llvm/tests/guest_runner.cc)
    target_link_libraries(llvm_guest_runner PRIVATE anc216core)
    add_test(NAME llvm_codegen COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/llvm/tests/codegen_test.py
        --llc ${ANC216_LLVM_LLC} --clang ${ANC216_LLVM_BIN}/clang
        --assembler $<TARGET_FILE:assembler> --runner $<TARGET_FILE:llvm_guest_runner>)
endif()
if(BUILD_TESTING)
    add_subdirectory(assembler/test)
    add_test(NAME legacy_assembler COMMAND assembler_test)
    set_tests_properties(legacy_assembler PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assembler/test" FAIL_REGULAR_EXPRESSION "failed;Different sizes")
    add_executable(cpu_test tests/cpu_test.cc)
    target_link_libraries(cpu_test PRIVATE anc216core)
    add_test(NAME cpu COMMAND cpu_test)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME toolchain COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/toolchain_test.py
        $<TARGET_FILE:assembler> $<TARGET_FILE:disassembler> $<TARGET_FILE:anc216emu> $<TARGET_FILE:cardreader>)
endif()

find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(OS_BUILD_OPTIONS)
if(BUILD_TESTING)
    list(APPEND OS_BUILD_OPTIONS --test-fixtures)
endif()
add_custom_target(os
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/os/build.py
        --assembler $<TARGET_FILE:assembler> --cardreader $<TARGET_FILE:cardreader> --output ${CMAKE_BINARY_DIR}/os ${OS_BUILD_OPTIONS}
    DEPENDS assembler cardreader
    COMMENT "Build ANC216 OS ROM and AVC64 charmap")
if(BUILD_TESTING)
    add_executable(os_test tests/os_test.cc)
    target_link_libraries(os_test PRIVATE anc216core)
    add_dependencies(os_test os)
    add_test(NAME os_boot COMMAND os_test ${CMAKE_BINARY_DIR}/os)
endif()

if(BUILD_TESTING)
    add_executable(host_input_test tests/host_input_test.cc)
    target_link_libraries(host_input_test PRIVATE anc216core)
    add_test(NAME host_input COMMAND host_input_test)
    set_tests_properties(host_input PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()

if(BUILD_TESTING)
    add_executable(debugger_test tests/debugger_test.cc)
    target_link_libraries(debugger_test PRIVATE anc216core)
    add_test(NAME debugger COMMAND debugger_test)
    if(UNIX)
        add_test(NAME debugger_console COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/debugger_console_test.py $<TARGET_FILE:anc216emu>)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(shell_test tests/shell_test.cc)
    target_link_libraries(shell_test PRIVATE anc216core)
    add_dependencies(shell_test os)
    add_test(NAME shell COMMAND shell_test ${CMAKE_BINARY_DIR}/os)
endif()
